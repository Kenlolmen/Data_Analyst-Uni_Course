import numpy as np

#zad1

def distp(X,C):
    LpunktowX = X.shape[0]
    LpunktowC = C.shape[0]

    o = np.zeros((LpunktowX,LpunktowC)) 

    for i in range(LpunktowX):
        for j in range(LpunktowC):
            r = X[i]-C[j] #roznica
            o[i,j] = np.sqrt(np.dot(r,r.T))
    return o

def distm(X,C):
    V = np.cov(X.T) 
    inv = np.linalg.pinv(V)

    LpunktowX = X.shape[0]
    LpunktowC = C.shape[0]

    o = np.zeros((LpunktowX,LpunktowC)) 
    for i in range(LpunktowX):
        for j in range(LpunktowC):
            r = X[i] - C[j]
            o[i, j] = np.sqrt(np.dot(np.dot(r, inv), r.T)) 
    return o

def ksrodki(X, k):
    n, d = X.shape
    C = X[np.random.choice(n, k, replace=False)]
    CX = np.zeros(n, dtype=int)

    for _ in range(100):
        d = distp(X, C)
        new_CX = np.argmin(d, axis=1)

        if np.array_equal(new_CX, CX): break
        CX = new_CX

        for i in range(k):
            mem = X[CX == i]
            if len(mem) > 0:
                C[i] = np.mean(mem, axis=0)

    return C, CX

X = np.array([[1, 2], [2, 3], [3, 4]])

import pandas as pd
import matplotlib.pyplot as plt
from sklearn.decomposition import PCA

# Wczytaj dane
df = pd.read_csv('autos.csv')

features1 = ['wheel-base', 'price', 'height', 'horsepower']
features2 = ['wheel-base', 'price']

for features in [features1, features2]:

    X = df[features].dropna().values 

    if len(features) > 3:
        pca = PCA(n_components=2)
        X_pca = pca.fit_transform(X)
        xlabel, ylabel = 'PCA 1', 'PCA 2'
    else:
        X_pca = X
        xlabel, ylabel = features[0], features[1]

    for k in [2, 4]:
        C, CX = ksrodki(X, k)

        plt.figure(figsize=(6, 5))
        plt.scatter(X_pca[:, 0], X_pca[:, 1], c=CX, cmap='tab10', s=30, alpha=1.0)
        plt.title(f'K-średnich z k={k} dla cech: {features}')
        plt.show()
